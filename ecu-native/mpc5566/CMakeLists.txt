cmake_minimum_required(VERSION 3.16)
project(CamaroDiagMpc5566HostTests LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_library(cd_flash STATIC flash_map.c flash_controller.c kernel_protocol.c)
target_compile_options(cd_flash PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(cd_flash_model STATIC tests/flash_model.c)
target_compile_options(cd_flash_model PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(cd_flash_model PUBLIC cd_flash)

foreach(test flash_map flash_controller kernel_protocol)
    add_executable(${test}_test tests/${test}_test.c)
    target_compile_options(${test}_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    target_link_libraries(${test}_test PRIVATE cd_flash cd_flash_model)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
